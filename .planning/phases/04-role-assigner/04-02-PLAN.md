parseCsv(text) — RESEARCH.md Pattern 5:
  Input: raw CSV string (may have CRLF or LF)
  Output: { rows: [{user, computerID, role, lineNum}] } on success
          { error: "Missing required columns: user, Computer_ID, role" } if header is wrong
  Header matching is case-insensitive (toLowerCase before indexOf)
  Empty lines are skipped

validateRows(rows, allowedRoles) — RESEARCH.md Pattern 5:
  Input: rows[] from parseCsv, allowedRoles string[]
  Output: { errors: [{rowIndex, field, msg}], warnings: [{rowIndex, msg}] }
  BLOCK errors: blank user → {field:'user', msg:'User is blank'}
                blank computerID → {field:'computerID', msg:'Computer_ID is blank'}
                role not in allowedRoles → {field:'role', msg:'Unknown role: "..."'}
  WARN only:  duplicate key (user|computerID|role) → {msg:'Duplicate row (will be ignored by upsert)'}

safeB64Encode(str) — RESEARCH.md Pattern 1 + Pitfall 1:
  In Node.js (test file): Buffer.from(str, 'utf8').toString('base64')
  In browser (OJO): btoa(unescape(encodeURIComponent(str)))
  The test file uses the Node.js path; the OJO uses the browser path.
  Round-trip: Buffer.from(safeB64Encode(s), 'base64').toString('utf8') === s

chunkString(s, size) — RESEARCH.md Pattern 1:
  Returns array of strings each <= size chars; last element may be shorter
  chunkString("abcde", 2) === ["ab", "cd", "e"]
  chunkString("", 3000) === []

buildChunkParams(b64) — RESEARCH.md Pattern 1:
  CHUNK_SIZE = 3000
  Returns multi-line string:
    parameter "chunk0"="<3000 chars>"
    parameter "chunk1"="<remaining>"
    parameter "chunkCount"="2"
  Each chunk value contains only B64 alphabet chars [A-Za-z0-9+/=]
  chunkCount is the last line

buildAssignActionXml(role, computerIds, actionScript, suffix, relevance) — RESEARCH.md Pattern 4:
  title = 'Dereso_' + role + ' Assign_' + suffix  (per D-09)
  XML contains <Title> with the above title
  XML contains one <ComputerID> child element per entry in computerIds array
  XML contains <Relevance> with the relevance string (DB-existence guard, verbatim from Dereso SWD.bes)
  XML contains <ActionScript MIMEType="application/x-Fixlet-Windows-Shell"> with CDATA-wrapped actionScript
  xmlEscape() applied to title and relevance before embedding
  cdataStart() = '<!' + '[CDATA['  cdataEnd() = ']]' + '>'

xmlEscape(s) — same as dispatcher.test.js line 31:
  & → &amp;  < → &lt;  > → &gt;  " → &quot;  ' → &apos;

xmlCDataSafe(s) — same as dispatcher.test.js line 41:
  ]]> → ]]]]><![CDATA[>
