Deliver `Dashboard_4_samples/RoleAwareness_Report.beswrpt` — a BigFix Web Report that
shows all role-aware fixlets in the Software Portal site and the actions generated from

## Task 2 — Async Data Loading via `Relevance()` API

Use the BigFix Web Reports `Relevance(queryString, callback)` JavaScript function.
Delimiter: `|` (single pipe), consistent with the sample.

**Two relevance strings** (verbatim from OQ-1 and OQ-2, adapted to `|` delimiter):

```javascript
var FIXLET_QUERY = '((item 0 of it as string & "|" & item 1 of it as string & "|" & item 2 of it as string & "|" & item 3 of it as string) of (id of it, name of it, name of site of it, (if (exists mime fields whose (name of it = "Associated_Roles") of it) then ((if (exists value whose (not(it = "")) of it) then (concatenation " | " of values of it) else "Nulled Role") of mime fields whose (name of it = "Associated_Roles") of it) else "General Access")) of ((bes fixlets) whose ((exists fixlet flag of it) and (exists site whose (name of it contains "Dereso Software Portal") of it))))';

var ACTION_QUERY = '((item 0 of it as string & "|" & item 1 of it as string & "|" & item 2 of it as string & "|" & (value of key "id" of json of item 3 of it) as string & "|" & (value of key "sitename" of json of item 3 of it) as string) of (id of it, name of it, (if (exists MIME Field whose (name of it = "Associated_Roles") of it) then (concatenation "" of values of MIME Fields whose (name of it = "Associated_Roles") of it) else "General Access") of it, (if (exists MIME Field whose (name of it = "SmartRole_Source") of it) then (concatenation "" of values of MIME Fields whose (name of it = "SmartRole_Source") of it) else ("{%22id%22:%22%22,%22sitename%22:%22%22}")) of it) of (bes actions whose ((exists offer flag of it) and (exists MIME Field whose (name of it = "Associated_Roles") of it) and (exists MIME Field whose (name of it = "SmartRole_Source") of it))))';
```

**`SearchForData()` — sequential async loads, fixlets first:**

```javascript
var fixletMap = {};  // keyed by fixletId + '|' + siteName

function SearchForData() {
  document.getElementById('msg1').innerHTML = 'Loading fixlets...';
  Relevance(FIXLET_QUERY, {
    success: function(results) {
      for (var i = 0; i < results.length; i++) {
        var f = results[i].split('|');
        var key = f[0] + '|' + f[2];   // id|site
        fixletMap[key] = {
          id:    f[0] || '',
          name:  f[1] || '',
          site:  f[2] || '',
          roles: (f[3] || '').split(' | ').map(function(r) { return r.trim(); }).filter(Boolean)
        };
      }
      document.getElementById('msg1').innerHTML = 'Loading actions...';
      loadActions();
    },
    failure: function(xhr, textStatus, errorThrown) {
      document.getElementById('msg1').innerHTML =
        'Error loading fixlets: ' + textStatus + ' — ' + errorThrown;
    }
  });
}

function loadActions() {
  Relevance(ACTION_QUERY, {
    success: function(results) {
      var DataList = [];
      for (var i = 0; i < results.length; i++) {
        var a = results[i].split('|');
        var srcId   = decodeURIComponent(a[3] || '');
        var srcSite = decodeURIComponent(a[4] || '');
        var fKey    = srcId + '|' + srcSite;
        var fixlet  = fixletMap[fKey] || { name: '(unknown)', site: srcSite, roles: [] };
        DataList.push([
          fixlet.name,                        // col 0 — grouping key (fixlet name)
          fixlet.roles.join(' | '),            // col 1 — fixlet roles
          a[0] || '',                          // col 2 — action ID
          a[1] || '',                          // col 3 — action name
          decodeURIComponent(a[2] || ''),      // col 4 — action role
          fixlet.site,                         // col 5 — site (hidden)
          srcId                                // col 6 — source fixlet ID (hidden)
        ]);
      }
      var z = $('#srch').DataTable();
      z.rows.add(DataList).draw();
      z.columns.adjust().draw();
      document.getElementById('msg1').innerHTML =
        DataList.length + ' action(s) loaded.';
    },
    failure: function(xhr, textStatus, errorThrown) {
      document.getElementById('msg1').innerHTML =
        'Error loading actions: ' + textStatus + ' — ' + errorThrown;
    }
  });
}
```

## Task 3 — DataTable Init + `drawCallback` Row Grouping

**Table initialised empty on `$(document).ready`**, populated by `loadActions()` callback.

**7 columns** (col 0 is the grouping key — fixlet name, rendered only in `.group` header rows):

| Col | Content | Visible | Notes |
|-----|---------|---------|-------|
| 0 | Fixlet Name | Hidden | Grouping key — `orderFixed: [0,'asc']` |
| 1 | Fixlet Roles | Visible | Rendered as `.role-badge` pills in group header |
| 2 | Action ID | Hidden (searchable) | |
| 3 | Action Name | Visible | |
| 4 | Action Role | Visible | `.role-badge.action` pill |
| 5 | Site | Hidden (searchable) | |
| 6 | Source Fixlet ID | Hidden | |

```javascript
$(document).ready(function() {
  $('#srch').DataTable({
    scrollY:    600,
    scrollX:    true,
    pagingType: 'full_numbers',
    stateSave:  true,
    order:      [[0, 'asc']],
    dom:        'QBlfrtip',
    buttons: [
      { extend: 'colvis', postfixButtons: ['colvisRestore'] },
      'excel', 'copy', 'pdf', 'csv'
    ],
    columnDefs: [
      { targets: [3, 4],          visible: true  },
      { targets: [0, 1, 2, 5, 6], visible: false, searchable: true }
    ],
    lengthMenu:    [[10, 25, 50, -1], [10, 25, 50, 'All']],
    displayLength: 25,
    drawCallback: function(settings) {
      var api      = this.api();
      var rowNodes = api.rows({ page: 'current' }).nodes();
      var rowIdx   = api.rows({ page: 'current' }).indexes();
      var last     = null;
      api.column(0, { page: 'current' }).data().each(function(fixletName, i) {
        if (last !== fixletName) {
          var roles = api.cell(rowIdx[i], 1).data() || '';
          var badges = roles.split(' | ').filter(Boolean)
            .map(function(r) { return '<span class="role-badge">' + escHtml(r.trim()) + '</span>'; })
            .join(' ');
          $(rowNodes).eq(i).before(
            '<tr class="group"><td colspan="5"><b>' + escHtml(fixletName) + '</b> &nbsp; ' + badges + '</td></tr>'
          );
          last = fixletName;
        }
      });
    }
  });

  SearchForData();
});

function escHtml(s) {
  return String(s)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}
```

`escHtml` is the XSS mitigation for relevance-sourced strings injected into the group header
HTML (FR-09 acceptance criterion).
